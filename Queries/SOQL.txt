===============================
Salesforce Student Management System
SOQL Queries
===============================

1. Retrieve All Student Records

SELECT Id, Name, Roll_Number__c, Email__c, Phone__c,
       Department__c, Year__c
FROM Student__c
ORDER BY Name;


--------------------------------------

2. Retrieve All Course Records

SELECT Id, Name, Course_Code__c, Credits__c
FROM Course__c
ORDER BY Name;


--------------------------------------

3. Retrieve All Enrollment Records

SELECT Id,
       Name,
       Student__r.Name,
       Course__r.Name,
       Status__c,
       Enrollment_Date__c
FROM Enrollment__c
ORDER BY CreatedDate DESC;


--------------------------------------

4. Retrieve Students from a Specific Department

SELECT Id, Name, Department__c
FROM Student__c
WHERE Department__c = 'CSE';


--------------------------------------

5. Retrieve Students in Final Year

SELECT Id, Name, Year__c
FROM Student__c
WHERE Year__c = 4;


--------------------------------------

6. Retrieve Courses with More Than 3 Credits

SELECT Id, Name, Credits__c
FROM Course__c
WHERE Credits__c > 3;
